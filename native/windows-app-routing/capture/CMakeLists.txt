cmake_minimum_required(VERSION 3.20)
project(EnvaraCapture LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(envara_capture_core STATIC capture_core.c)
target_include_directories(envara_capture_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(envara_capture_core PRIVATE /W4)
else()
  target_compile_options(envara_capture_core PRIVATE -Wall -Wextra -Werror)
endif()
option(ENVARA_CAPTURE_BUILD_TESTS "Build portable tuple/identity/NAT lease tests" ON)
if(ENVARA_CAPTURE_BUILD_TESTS)
  enable_testing()
  add_executable(envara_capture_core_test tests/core_test.c)
  target_link_libraries(envara_capture_core_test PRIVATE envara_capture_core)
  add_test(NAME capture_core COMMAND envara_capture_core_test)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Capture ABI and official WinDivert SDK currently target Windows x64 only")
  endif()
  set(WINDIVERT_ROOT "" CACHE PATH "Official unchanged WinDivert 2.2.2-A SDK root")
  if(NOT EXISTS "${WINDIVERT_ROOT}/x64/WinDivert.lib")
    message(FATAL_ERROR "Set WINDIVERT_ROOT to verified official WinDivert SDK")
  endif()
  add_library(EnvaraCapture SHARED capture_windows.c capture_core.c EnvaraCapture.def)
  target_compile_definitions(EnvaraCapture PRIVATE ENVARA_CAPTURE_BUILD WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(EnvaraCapture PRIVATE "${WINDIVERT_ROOT}/x64/WinDivert.lib" ws2_32 iphlpapi)
  set_target_properties(EnvaraCapture PROPERTIES PREFIX "" OUTPUT_NAME "EnvaraCapture")
  if(MSVC)
    target_compile_options(EnvaraCapture PRIVATE /W4 /GS)
    target_link_options(EnvaraCapture PRIVATE /DYNAMICBASE /NXCOMPAT /GUARD:CF)
  else()
    target_compile_options(EnvaraCapture PRIVATE -Wall -Wextra -Werror)
    target_link_options(EnvaraCapture PRIVATE -Wl,--dynamicbase,--nxcompat)
  endif()
endif()
